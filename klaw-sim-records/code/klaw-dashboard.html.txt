// ── Supabase fetch
async function _sb(path) {
  const res = await fetch(SB_URL + '/rest/v1/' + path, {
    headers: { 'apikey': SB_KEY, 'Authorization': 'Bearer ' + SB_KEY, 'Content-Type': 'application/json' }
  });
  if (!res.ok) throw new Error('SB ' + res.status);
  return res.json();
}

// ── 필터 쿼리 빌더
function _buildFilter(extra='') {
  const parts = [];
  if (_typeFilter) parts.push(`case_type=eq.${encodeURIComponent(_typeFilter)}`);
  const df = document.getElementById('dateFrom')?.value;
  const dt = document.getElementById('dateTo')?.value;
  if (df) parts.push(`created_at=gte.${df}T00:00:00+09:00`);
  if (dt) parts.push(`created_at=lte.${dt}T23:59:59+09:00`);
  const q = parts.join('&');
  return (q ? q + '&' : '') + extra;
}

// ── KPI
async function loadKPIs() {
  try {
    const q  = _buildFilter(`select=id,verdict,verdict_prob,confidence,risk_level,created_at`);
    const rows = await _sb(`${SB_TABLE}?${q}&order=created_at.desc&limit=1000`);

    const total = rows.length;
    const today = new Date().toISOString().slice(0,10);
    const todayCnt = rows.filter(r => r.created_at?.slice(0,10) === today).length;
    const probs = rows.filter(r => r.verdict_prob != null).map(r => r.verdict_prob);
    const avgProb = probs.length ? (probs.reduce((a,b)=>a+b,0)/probs.length).toFixed(1) : '—';
    const wins = rows.filter(r => ['원고승','무죄','일부인용'].includes(r.verdict)).length;
    const winRate = total ? Math.round(wins/total*100) + '%' : '—';

    document.getElementById('kpiTotal').textContent   = total.toLocaleString();

    document.getElementById('kpiAvgProb').textContent = avgProb !== '—' ? avgProb + '%' : '—';
    document.getElementById('kpiWinRate').textContent  = winRate;
    document.getElementById('chartTotalBadge').textContent = total + '건';

    // pass rows to charts

// ── Timeline (14일)
function renderTimeline(rows) {
  const svg = document.getElementById('timelineSvg');
  svg.innerHTML = '';
  if (!rows.length) return;

  const days = [];
  for (let i=13; i>=0; i--) {
    const d = new Date(Date.now() - i*86400000);
    days.push(d.toISOString().slice(0,10));
  }
  const counts = days.map(d => rows.filter(r => r.created_at?.slice(0,10) === d).length);
  const maxC = Math.max(...counts, 1);

  const W = 600, H = 140, pad = { left:30, right:10, top:10, bottom:25 };
  const bW = (W - pad.left - pad.right) / days.length;

  // bars
  days.forEach((d, i) => {
    const bH = ((counts[i] / maxC) * (H - pad.top - pad.bottom)) || 2;
    const x = pad.left + i * bW;
    const y = H - pad.bottom - bH;
    const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
    rect.setAttribute('x', x + 2);
    rect.setAttribute('y', y);
    rect.setAttribute('width', bW - 4);
    rect.setAttribute('height', bH);
    rect.setAttribute('rx', '3');
    rect.setAttribute('fill', counts[i] ? '#3ecf8e' : '#242427');
    rect.setAttribute('opacity', '0.85');
    svg.appendChild(rect);

    if (i % 2 === 0 || i === days.length-1) {
      const lbl = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      lbl.setAttribute('x', x + bW/2);
      lbl.setAttribute('y', H - 8);
      lbl.setAttribute('text-anchor', 'middle');
      lbl.setAttribute('class', 'bar-label');
      lbl.textContent = d.slice(5); // MM-DD
      svg.appendChild(lbl);
    }

    if (counts[i]) {
      const val = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      val.setAttribute('x', x + bW/2);
      val.setAttribute('y', y - 3);
      val.setAttribute('text-anchor', 'middle');
      val.setAttribute('class', 'bar-val');
      val.textContent = counts[i];
      svg.appendChild(val);
    }
  });
}

// ── Donut chart
const COLORS = ['#3ecf8e','#6e56cf','#3b82f6','#f5a623','#f04040','#14b8a6','#f472b6','#a78bfa','#34d399'];
function renderDonut(rows) {
  const typeCount = {};
  rows.forEach(r => { typeCount[r.case_type] = (typeCount[r.case_type]||0) + 1; });
  const sorted = Object.entries(typeCount).sort((a,b)=>b[1]-a[1]);
  const total = rows.length || 1;

  const svg = document.getElementById('donutSvg');
  // clear old paths
  [...svg.querySelectorAll('path,circle')].forEach(el => el.remove());

  const cx=60, cy=60, r=45, ir=28;
  let angle = -Math.PI/2;

  sorted.forEach(([type, cnt], i) => {
    const sweep = (cnt/total) * Math.PI * 2;
    const x1 = cx + r*Math.cos(angle), y1 = cy + r*Math.sin(angle);
    const x2 = cx + r*Math.cos(angle+sweep), y2 = cy + r*Math.sin(angle+sweep);
    const ix1 = cx + ir*Math.cos(angle), iy1 = cy + ir*Math.sin(angle);
    const ix2 = cx + ir*Math.cos(angle+sweep), iy2 = cy + ir*Math.sin(angle+sweep);
    const large = sweep > Math.PI ? 1 : 0;
    const path = document.createElementNS('http://www.w3.org/2000/svg','path');
    path.setAttribute('d', `M${ix1},${iy1} L${x1},${y1} A${r},${r} 0 ${large} 1 ${x2},${y2} L${ix2},${iy2} A${ir},${ir} 0 ${large} 0 ${ix1},${iy1} Z`);
    path.setAttribute('fill', COLORS[i % COLORS.length]);
    svg.insertBefore(path, svg.querySelector('text'));
    angle += sweep;
  });

  document.getElementById('donutCenter').textContent = total + '건';

  const legend = document.getElementById('donutLegend');
  legend.innerHTML = sorted.slice(0,6).map(([type,cnt],i) =>
    `<div class="legend-item">
      <div class="legend-dot" style="background:${COLORS[i%COLORS.length]}"></div>
      <div class="legend-label">${type}</div>
      <div class="legend-val">${cnt}</div>
    </div>`
  ).join('');
}

// ── Verdict breakdown
function renderVerdictBreakdown(rows) {
  const types = ['원고승','일부인용','피고승','기각','각하','화해권고','무죄','유죄','집행유예'];
  const colors = { '원고승':'#3ecf8e','일부인용':'#f5a623','피고승':'#f04040','기각':'#f04040','각하':'#6b6b78','화해권고':'#6e56cf','무죄':'#3ecf8e','유죄':'#f04040','집행유예':'#f5a623' };
  const cnt = {};
  rows.forEach(r => r.verdict && (cnt[r.verdict]=(cnt[r.verdict]||0)+1));
  const max = Math.max(...Object.values(cnt),1);
  const total = rows.length || 1;

  const filtered = types.filter(t => cnt[t]);
  const vl = document.getElementById('verdictList');
  vl.innerHTML = filtered.length ? filtered.map(t =>
    `<div class="verdict-item">
      <div class="verdict-label">${t}</div>
      <div class="verdict-bar-bg"><div class="verdict-bar-fill" style="width:${Math.round((cnt[t]||0)/max*100)}%;background:${colors[t]||'var(--primary)'}"></div></div>
      <div class="verdict-count">${cnt[t]||0}</div>
    </div>`
  ).join('') : '<div style="color:var(--text-3);font-size:12px;padding:8px;">데이터 없음</div>';
}

// ── Table
async function loadTable() {
  const tbody = document.getElementById('simTable');
  tbody.innerHTML = '<tr><td colspan="7" style="text-align:center;padding:40px;color:var(--text-3);">로딩 중…</td></tr>';
  try {
    const q = _buildFilter(`select=id,case_type,case_subtype,case_summary,verdict,verdict_prob,risk_level,confidence,created_at`);
    const rows = await _sb(`${SB_TABLE}?${q}&order=created_at.desc&limit=${PAGE_SIZE}&offset=${_page*PAGE_SIZE}`);

    if (!rows.length) {
      tbody.innerHTML = '<tr><td colspan="7" style="text-align:center;padding:40px;color:var(--text-3);">결과 없음</td></tr>';
      document.getElementById('tableCount').textContent = '0건';
      return;
    }

    const verdictBadge = v => {
      const m = {'원고승':'badge-win','무죄':'badge-win','피고승':'badge-lose','기각':'badge-lose','각하':'badge-lose','일부인용':'badge-part','화해권고':'badge-part','유죄':'badge-lose','집행유예':'badge-part'};
      return m[v] || 'badge-pend';
    };
    const riskBadge = r => ({'CRITICAL':'badge-crit','HIGH':'badge-high','MED':'badge-med','LOW':'badge-low'}[r] || 'badge-pend');
    const typeIcon  = t => ({'민사':'📝','형사':'🔒','가사':'👨‍👩‍👧','행정':'🏛️','노동':'👷','국제':'🌏','헌법':'🇰🇷','지재':'💡','세무':'💰'}[t]||'⚖️');

    tbody.innerHTML = rows.map(r => {
      const prob = r.verdict_prob ?? null;
      const probHtml = prob != null
        ? `<div class="prob-inline"><div class="prob-bar-sm"><div class="prob-fill-sm" style="width:${prob}%"></div></div><span class="prob-text">${prob}%</span></div>`
        : '<span style="color:var(--text-3)">—</span>';
      const date = r.created_at ? new Date(r.created_at).toLocaleDateString('ko-KR',{timeZone:'Asia/Seoul'}) : '—';
      const summary = r.case_summary ? r.case_summary.slice(0,40) + (r.case_summary.length>40?'…':'') : '—';
      return `<tr>
        <td class="td-id">${r.id}</td>
        <td class="td-type">${typeIcon(r.case_type)} ${r.case_type}${r.case_subtype ? '<br><span style="font-size:10px;color:var(--text-3);">'+r.case_subtype+'</span>' : ''}</td>
        <td style="max-width:200px;color:var(--text-2);">${summary}</td>
        <td>${r.verdict ? `<span class="badge ${verdictBadge(r.verdict)}">${r.verdict}</span>` : '<span style="color:var(--text-3)">—</span>'}</td>
        <td>${probHtml}</td>
        <td>${r.risk_level ? `<span class="badge ${riskBadge(r.risk_level)}">${r.risk_level}</span>` : '—'}</td>
        <td style="color:var(--text-3);font-size:12px;">${date}</td>
      </tr>`;
    }).join('');

    const cnt = rows.length;
    document.getElementById('tableCount').textContent = `${_page*PAGE_SIZE+1}–${_page*PAGE_SIZE+cnt}건 표시`;
    document.getElementById('pageInfo').textContent    = `${_page+1} 페이지`;
    document.getElementById('btnPrev').disabled = _page === 0;
    document.getElementById('btnNext').disabled = cnt < PAGE_SIZE;
  } catch(e) {
    tbody.innerHTML = `<tr><td colspan="7" style="text-align:center;padding:30px;color:var(--text-3);">${e.message}</td></tr>`;
  }
}

// ── CSV export
async function exportCSV() {
  try {
    const q = _buildFilter('select=id,case_type,case_subtype,case_summary,verdict,verdict_prob,risk_level,confidence,created_at,source');
    const rows = await _sb(`${SB_TABLE}?${q}&order=created_at.desc&limit=5000`);

    const csv = [headers.join(','), ...rows.map(r => [
      r.id, r.case_type, r.case_subtype||'', `"${(r.case_summary||'').replace(/"/g,'""')}"`,
      r.verdict||'', r.verdict_prob??'', r.risk_level||'', r.confidence??'',
      r.created_at?.slice(0,10)||'', r.source||''
    ].join(','))].join('\n');
    const a = Object.assign(document.createElement('a'), {
      href: 'data:text/csv;charset=utf-8,\uFEFF' + encodeURIComponent(csv),
      download: `klaw-export-${new Date().toISOString().slice(0,10)}.csv`
    });
    a.click();
  } catch(e) { alert('내보내기 실패: ' + e.message); }
}
