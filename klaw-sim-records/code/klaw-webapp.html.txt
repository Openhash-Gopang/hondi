// ── DeepSeek 호출 (직접 → Worker fallback)
async function _deepseekChat(messages, maxTokens=2000) {
  const payload = { model: 'deepseek-chat', messages, max_tokens: maxTokens, temperature: 0.3 };
  try {
    const r = await fetch('https://api.deepseek.com/v1/chat/completions', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${DEEPSEEK_KEY}` },
      body: JSON.stringify(payload),
      signal: AbortSignal.timeout(40000),
    });
    if (r.ok) return await r.json();
  } catch(e) { console.warn('[DeepSeek] 직접 실패:', e.message); }
  const r2 = await fetch(`${PROXY_URL}/deepseek`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload),
  });
  if (!r2.ok) throw new Error('DeepSeek proxy ' + r2.status);
  return r2.json();
}

async function runKLawSimulation(caseType, subtype, summary, laws, party) {
  const systemPrompt = `당신은 대한민국 최고법원 판례를 학습한 K-Law v15 AI 판결 예측 시스템입니다.

{
  "verdict": "원고승|피고승|일부인용|기각|각하|화해권고|무죄|유죄|집행유예",

  ],
  "reasoning": "판결 이유 3~5문장. 유사 판례 인용 포함.",
  "risk_level": "LOW|MED|HIGH|CRITICAL",
  "confidence": 신뢰도숫자(0~100)
}

판례 검색 규칙:
- 유사 대법원·고등법원 판결 2~3건 reasoning에 인용
- 사건 유형에 따른 핵심 법리 적용
- 승소 확률은 판례 통계 기반 현실적 수치`;

  const userMsg = `사건 유형: ${caseType}${subtype ? ' > ' + subtype : ''}
의뢰인 입장: ${party}
관련 법령: ${laws.length > 0 ? laws.join(', ') : '미지정'}

사건 개요:
${summary}

위 사건에 대해 K-Law v15 판결 예측을 JSON으로 출력하라.`;

  const data = await _deepseekChat([
    { role: 'system', content: systemPrompt },
    { role: 'user',   content: userMsg },
  ], 2000);

  const raw = data.choices?.[0]?.message?.content || '{}';
  try {
    return JSON.parse(raw.replace(/```json|```/g,'').trim());

// ── 시뮬레이션 시작
async function startSimulation() {
  const type    = document.getElementById('caseTypeSelect').value;
  const subtype = document.getElementById('caseSubtypeSelect').value;
  const summary = document.getElementById('caseSummary').value.trim();
  const party   = document.getElementById('partySelect').value;
  const laws    = [...document.querySelectorAll('.law-chip.active')].map(el => el.textContent);

  if (!type)    { alert('사건 유형을 선택하세요.'); return; }
  if (!summary) { alert('사건 개요를 입력하세요.'); return; }

  navTo('screen-analysis');
  const setStep = (n, done=false) => {
    for (let i=1; i<=5; i++) {
      const dot = document.getElementById(`dot${i}`);
      const sp  = document.getElementById(`sp${i}`);
      const ck  = document.getElementById(`ck${i}`);
      if (!dot) continue;
      document.getElementById(`step${i}`)?.classList.add('visible');
      if (i < n)      { dot.className='ai-step-dot done'; sp.style.display='none'; ck.style.display='block'; }
      else if (i===n) { dot.className='ai-step-dot active'; sp.style.display='block'; ck.style.display='none'; }
      else            { dot.className='ai-step-dot'; sp.style.display='none'; ck.style.display='none'; }
    }
  };

  let result = null;
  const simId = 'SIM-2026-' + (Math.floor(Math.random()*90000)+10000);

  try {
    setStep(1); await _sleep(400);
    setStep(2); await _sleep(600);
    setStep(3); await _sleep(400);
    setStep(4);

    result = await runKLawSimulation(type, subtype, summary, laws, party);

    setStep(5);

    // Supabase 저장
    const payload = {
      id:           simId,
      created_at:   _nowKST(),
      case_type:    type,
      case_subtype: subtype || null,
      case_summary: summary,
      reporter:     localStorage.getItem('gopang_phone') || null,
      source:       GopangWidget.isWidget ? 'gopang' : 'webapp',
      verdict:      result.verdict || null,
      verdict_prob: result.verdict_prob ?? null,
      damages:      result.damages ?? null,
      sentence:     result.sentence || null,
      key_issues:   result.key_issues || [],
      applicable_laws: result.applicable_laws || [],
      reasoning:    result.reasoning || null,
      risk_level:   result.risk_level || 'MED',
      confidence:   result.confidence ?? null,
      status:       '완료',
      analysis:     result,
    };
    try { await _sbFetch(SB_TABLE, 'POST', payload); } catch(e) { console.warn('[SB저장실패]', e.message); }

    await _sleep(500);
    showResult(simId, type, result);

    // GWP 복귀
    if (GopangWidget.isWidget) {
      setTimeout(() => {
        window.__gwpInstance?.done({
          summary: `${type} 판결 시뮬레이션: ${result.verdict} (승소확률 ${result.verdict_prob}%)`,
          pdvData: {
            when: new Date().toISOString(),
            what: `K-Law 판결 예측 — ${simId}`,
            how:  'text',
            why:  `${type} 법률 시뮬레이션 요청`,
            data: { simId, caseType: type, verdict: result.verdict, prob: result.verdict_prob },
          },
        });
      }, 3000);
    }

  } catch(e) {
    console.error('[K-Law 오류]', e);
    result = { verdict:'분석불가', verdict_prob:0, reasoning:'AI 분석 중 오류가 발생했습니다: ' + e.message, risk_level:'MED', confidence:0, key_issues:[], applicable_laws:[] };
    showResult(simId, type, result);
  }
}

function _sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

    '무죄':     { cls:'win',  icon:'✅', sub:'무죄 판결' },
    '피고승':   { cls:'lose', icon:'❌', sub:'피고 승소 (원고 패소)' },
    '기각':     { cls:'lose', icon:'🚫', sub:'청구 기각' },
    '각하':     { cls:'lose', icon:'⛔', sub:'소각하' },
    '유죄':     { cls:'lose', icon:'⚠️', sub:'유죄 판결' },
    '일부인용': { cls:'part', icon:'⚖️', sub:'일부 인용' },
    '집행유예': { cls:'part', icon:'🔔', sub:'집행유예' },
    '화해권고': { cls:'part', icon:'🤝', sub:'화해 권고' },
  };
  const vs = verdictStyle[r.verdict] || { cls:'pend', icon:'⚖️', sub:'분석 완료' };

  const hdr = document.getElementById('resultHeader');
  hdr.className = 'result-header ' + vs.cls;
  document.getElementById('resultIcon').textContent    = vs.icon;
  document.getElementById('resultVerdict').textContent = r.verdict || '—';
  document.getElementById('resultSubtitle').textContent = vs.sub;
  document.getElementById('resultId').textContent      = simId;

  const prob = r.verdict_prob ?? 0;
  document.getElementById('probLabel').textContent = `${prob}%`;
  setTimeout(() => { document.getElementById('probFill').style.width = prob + '%'; }, 100);

  document.getElementById('resType').textContent    = caseType;
  document.getElementById('resVerdict').textContent = r.verdict || '—';
  document.getElementById('resDamages').textContent = r.damages ? `₩${Number(r.damages).toLocaleString()}` : (r.sentence || '해당없음');
  document.getElementById('resConf').textContent    = r.confidence ? `${r.confidence}%` : '—';

  // 쟁점
  const il = document.getElementById('issueList');
  il.innerHTML = (r.key_issues || []).map((iss, i) =>
    `<li class="issue-item"><div class="issue-num">${i+1}</div><span style="font-size:12px;">${iss}</span></li>`
  ).join('') || '<li class="issue-item">쟁점 정보 없음</li>';

  // 법령
  const ll = document.getElementById('lawList');
  ll.innerHTML = (r.applicable_laws || []).map(l =>
    `<div class="law-item"><div class="law-item-code">${l.law || ''}</div><div class="law-item-desc">${l.desc || ''}</div></div>`
  ).join('') || '<div style="font-size:12px;color:var(--muted);">법령 정보 없음</div>';

  document.getElementById('reasoningText').textContent = r.reasoning || '—';
}

// ── 홈 통계 로드
async function _loadStats() {
  try {
    const today = new Date().toISOString().slice(0,10);
    const [all, tod] = await Promise.all([
      _sbFetch(`${SB_TABLE}?select=id&limit=1&head=true`).catch(() => null),
      _sbFetch(`${SB_TABLE}?select=id&created_at=gte.${today}T00:00:00+09:00`).catch(() => []),
    ]);
    // total via count header는 head=true 필요 → 간단히 배열 조회
    const allRows = await _sbFetch(`${SB_TABLE}?select=id`).catch(() => []);
    document.getElementById('statTotal').textContent = allRows.length || '0';
    document.getElementById('statToday').textContent = (Array.isArray(tod) ? tod.length : 0);
  } catch(e) { console.warn('[stats]', e.message); }

  // 최근 목록
  try {
    const rows = await _sbFetch(`${SB_TABLE}?select=id,case_type,case_summary,verdict,verdict_prob,created_at&order=created_at.desc&limit=5`);
    const rl = document.getElementById('recentList');
    if (!rows.length) { rl.innerHTML = '<div style="text-align:center;padding:30px;color:var(--muted);">아직 시뮬레이션이 없습니다.</div>'; return; }
    const typeIcon = t => ({'민사':'📝','형사':'🔒','가사':'👨‍👩‍👧','행정':'🏛️','노동':'👷','국제':'🌏','헌법':'🇰🇷','지재':'💡','세무':'💰'}[t] || '⚖️');
    const verdictBadge = v => {
      const m = {'원고승':'badge-win','무죄':'badge-win','피고승':'badge-lose','기각':'badge-lose','각하':'badge-lose','일부인용':'badge-partial','화해권고':'badge-partial','유죄':'badge-lose','집행유예':'badge-partial'};
      return m[v] || 'badge-pend';
    };
    rl.innerHTML = rows.map(r => `
      <div class="recent-item">
        <div class="recent-icon">${typeIcon(r.case_type)}</div>
        <div class="recent-info">
          <div class="recent-title">${r.case_type}${r.case_summary ? ' — ' + r.case_summary.slice(0,30) + '…' : ''}</div>
          <div class="recent-sub">${r.id} · ${r.created_at ? new Date(r.created_at).toLocaleDateString('ko-KR',{timeZone:'Asia/Seoul'}) : ''}</div>
        </div>
        <span class="recent-badge ${verdictBadge(r.verdict)}">${r.verdict || '처리중'}</span>
      </div>`).join('');
  } catch(e) { console.warn('[recentList]', e.message); }
}

// ── 기록 화면
let _histRows = [];
async function _loadHistory(filter) {
  if (filter !== undefined) _histFilter = filter;
  const list = document.getElementById('historyList');
  list.innerHTML = '<div style="text-align:center;padding:30px;color:var(--muted);">불러오는 중…</div>';
  try {
    let q = `${SB_TABLE}?select=id,case_type,case_subtype,case_summary,verdict,verdict_prob,created_at,risk_level&order=created_at.desc&limit=50`;
    if (_histFilter) q += `&case_type=eq.${encodeURIComponent(_histFilter)}`;
    _histRows = await _sbFetch(q);
    _renderHistory(_histRows);
  } catch(e) {
    list.innerHTML = `<div style="text-align:center;padding:30px;color:var(--muted);">불러오기 실패<br><small>${e.message}</small></div>`;
  }
}

function filterHistory(el, filter) {
  document.querySelectorAll('.hf-chip').forEach(c => c.classList.remove('active'));
  el.classList.add('active');
  _loadHistory(filter);
}

function _renderHistory(rows) {

  const typeIcon = t => ({'민사':'📝','형사':'🔒','가사':'👨‍👩‍👧','행정':'🏛️','노동':'👷','국제':'🌏','헌법':'🇰🇷','지재':'💡','세무':'💰'}[t] || '⚖️');
  const verdictBadge = v => {
    const m = {'원고승':'badge-win','무죄':'badge-win','피고승':'badge-lose','기각':'badge-lose','각하':'badge-lose','일부인용':'badge-partial','화해권고':'badge-partial','유죄':'badge-lose','집행유예':'badge-partial'};

  };
  list.innerHTML = rows.map(r => `
    <div class="history-card">
      <div class="hc-top">
        <div class="hc-type">${typeIcon(r.case_type)} ${r.case_type}${r.case_subtype ? ' · ' + r.case_subtype : ''}</div>
        <div class="hc-date">${r.created_at ? new Date(r.created_at).toLocaleDateString('ko-KR',{timeZone:'Asia/Seoul'}) : ''}</div>

      </div>
    </div>`).join('');
}
